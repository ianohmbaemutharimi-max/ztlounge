    function getPageId() {
        if (window.PageIdentity && window.PageIdentity.getCanonicalPageId) {
            var pageId = window.PageIdentity.getCanonicalPageId();
            console.log('[LOOP49][ContentLoader] getPageId() via PageIdentity:', pageId);
            return pageId;
        }
        // Fallback: EXACT basename mapping ONLY (never substrings, never
        // directory/hosting-path scanning). Mirrors PageIdentity.FILENAME_MAP
        // so unwired pages degrade without guessing page identity.
        if (window.CMSReport) {
            window.CMSReport.warn('ContentLoader', 'getPageId', 'MISSING_DEPENDENCY', 'window.PageIdentity (using exact-basename fallback)');
        }
        var segments = String(window.location.pathname || '').split('/');
        var file = (segments[segments.length - 1] || '').split('?')[0].split('#')[0].toLowerCase();
        var basenameMap = {
            'index.html': 'home',
            '': 'home',
            'restaurant.html': 'restaurant',
            'restaurant-menu.html': 'restaurant-menu',
            'lounge.html': 'lounge',
            'vip.html': 'vip',
            'events.html': 'events',
            'conference.html': 'conference',
            'autospa.html': 'autospa',
            'barbershop.html': 'autospa',  // Barbershop redirects to Auto Spa (consolidated)
            'grooming-care.html': 'grooming-care',  // Grooming & Care consolidated page
            'gallery.html': 'gallery',
            'contact.html': 'contact'
        };
        var fallbackPageId = Object.prototype.hasOwnProperty.call(basenameMap, file) ? basenameMap[file] : null;
        console.log('[LOOP49][ContentLoader] getPageId() fallback for file "' + file + '":', fallbackPageId);
        return fallbackPageId;
    }
