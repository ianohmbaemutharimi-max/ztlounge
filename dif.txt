+    function formatOpeningHours(hours, key) {
+        if (!hours) return '';
+        var direct = asHoursString(hours);
+        if (direct) return direct;
+        if (typeof hours !== 'object') return '';
+
+        var keyed = key ? asHoursString(hours[key]) : '';
+        if (keyed) return keyed;
+        var fallback = asHoursString(hours['default']);
+        if (fallback) return fallback;
+
+        var days = HOURS_DAYS.map(function (day) {
+            return { day: day, range: asHoursString(hours[day]) };
+        });
+        if (!days.some(function (d) { return d.range; })) return '';
+
+        // Group consecutive days that share the same range.
+        var groups = [];
+        days.forEach(function (d) {
+            var last = groups[groups.length - 1];
+            if (last && last.range === d.range) { last.to = d.day; return; }
+            groups.push({ from: d.day, to: d.day, range: d.range });
+        });
