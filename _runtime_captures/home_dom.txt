                    if (isActive) {

                // === FOCUS TRAPPING ===
                function getFocusableElements(container) {
                    return container.querySelectorAll(
                        'a[href], button:not([disabled]), input:not([disabled]), select:not([disabled]), textarea:not([disabled]), [tabindex="0"]'
                    );
                }

                function trapFocus(container) {
                    if (focusTrapActive) return;
                    
                    var focusableElements = getFocusableElements(container);
                    if (focusableElements.length === 0) return;
                    
                    var firstFocusable = focusableElements[0];
                    var lastFocusable = focusableElements[focusableElements.length - 1];
                    focusTrapActive = true;
                    
                    firstFocusable.focus();
                    
                    focusTrapHandler = function(e) {
                        if (e.key !== 'Tab') return;
                        if (e.shiftKey) {
                            if (document.activeElement === firstFocusable) {
                                // e.preventDefault();
                                lastFocusable.focus();
                            }
                        } else {
                            if (document.activeElement === lastFocusable) {
                                // e.preventDefault();
                                firstFocusable.focus();
                            }
                        }
                    };
                    
                    document.addEventListener('keydown', focusTrapHandler);
                }

                function releaseFocus() {
                    if (focusTrapHandler) {
                        document.removeEventListener('keydown', focusTrapHandler);
                        focusTrapHandler = null;
                    }
                    focusTrapActive = false;
                }

                // === MOBILE DROPDOWN TOGGLE ===
                var dropdownParents = navMenu.querySelectorAll('.nav-list > li > a.dropdown-parent');
                dropdownParents.forEach(function(link) {
                    link.addEventListener('click', function(e) {
                        if (window.innerWidth <= 768) {
                            // e.preventDefault();
                            var parentLi = this.parentElement;
                            var isExpanded = parentLi.getAttribute('aria-expanded') === 'true';
                            
                            // Close other dropdowns
                            var activeItems = navMenu.querySelectorAll('.nav-list > li.active-dropdown');
                            activeItems.forEach(function(el) {
                                if (el !== parentLi) {
                                    el.classList.remove('active-dropdown');
                                    el.setAttribute('aria-expanded', 'false');
                                }
                            });
                            
                            var newExpanded = !isExpanded;
                            parentLi.classList.toggle('active-dropdown', newExpanded);
                            parentLi.setAttribute('aria-expanded', String(newExpanded));
                        }
                    });

                    // === KEYBOARD NAVIGATION FOR DROPDOWNS ===
                    link.addEventListener('keydown', function(e) {
                        if (e.key === 'ArrowDown' || e.key === 'ArrowUp') {
                            // e.preventDefault();
                            var menu = this.parentElement.querySelector('.dropdown-menu');
                            if (menu) {
                                // Ensure dropdown is open on desktop
                                var parentLi = this.parentElement;
                                if (window.innerWidth > 768) {
                                    parentLi.setAttribute('aria-expanded', 'true');
                                }
                                var firstItem = menu.querySelector('a');
                                if (firstItem) {
                                    firstItem.focus();
                                }
                            }
                        }
                    });
                });

                // Arrow key navigation within dropdown
                var dropdownMenus = document.querySelectorAll('.dropdown-menu');
                dropdownMenus.forEach(function(menu) {
                    var items = menu.querySelectorAll('a');
                    items.forEach(function(item, index) {
                        item.addEventListener('keydown', function(e) {
                            if (e.key === 'ArrowDown') {
                                // e.preventDefault();
                                var next = items[index + 1] || items[0];
                                next.focus();
                            } else if (e.key === 'ArrowUp') {
                                // e.preventDefault();
                                var prev = items[index - 1] || items[items.length - 1];
                                prev.focus();
                            } else if (e.key === 'Escape') {
                                // e.preventDefault();
                                var parentLink = this.closest('li')?.parentElement?.previousElementSibling;
                                if (parentLink && parentLink.classList.contains('dropdown-parent')) {
                                    parentLink.focus();
                                    var parentLi = parentLink.parentElement;
                                    if (window.innerWidth <= 768) {
                                        parentLi.classList.remove('active-dropdown');
                                        parentLi.setAttribute('aria-expanded', 'false');
                                    } else {
                                        parentLi.setAttribute('aria-expanded', 'false');
                                    }
                                }
                            }
                        });
                    });
                });

                // Close menu on resize to desktop
                window.addEventListener('resize', function() {
                    if (window.innerWidth > 768 && navMenu.classList.contains('active')) {
                        // // closeMobileMenu();
                    }
                });
            }

            // === HERO SLIDER ===
            // Live accessors - the canonical renderer (content-loader.js
            // populateHero) generates hero slide/dot DOM from hero_slides after
            // this block runs, so no stale NodeList snapshot may be held.
            function getHeroSlides() { return document.querySelectorAll('.hero-slider .hero-slide'); }
            function getHeroDots() { return document.querySelectorAll('.slider-dots .dot'); }
            var heroCurrent = 0;
            var heroInterval;

            function goToHero(index) {
                var heroSlides = getHeroSlides();
                var heroDots = getHeroDots();
                if (!heroSlides.length) return;
                if (index < 0) index = heroSlides.length - 1;
                if (index >= heroSlides.length) index = 0;
                heroSlides.forEach(function(slide, i) {
                    var isActive = i === index;
                    slide.classList.toggle('active', isActive);
                    slide.setAttribute('aria-hidden', String(!isActive));
                    var interactiveElements = slide.querySelectorAll('a, button');
                    interactiveElements.forEach(function(el) {
                        el.tabIndex = isActive ? 0 : -1;
                    });
                });
                heroDots.forEach(function(dot, i) {
                    var isActive = i === index;
                    dot.classList.toggle('active', isActive);
                    dot.setAttribute('aria-selected', String(isActive));
                    dot.tabIndex = isActive ? 0 : -1;
                });
                heroCurrent = index;
            }

            function nextHero() { goToHero(heroCurrent + 1); }
            function prevHero() { goToHero(heroCurrent - 1); }

            var prevSlideBtn = document.getElementById('prevSlide');
            var nextSlideBtn = document.getElementById('nextSlide');
            
            if (prevSlideBtn) prevSlideBtn.addEventListener('click', function() { prevHero(); resetHeroAutoplay(); });
            if (nextSlideBtn) nextSlideBtn.addEventListener('click', function() { nextHero(); resetHeroAutoplay(); });
            
            function bindHeroDots() {
                getHeroDots().forEach(function(dot, index) {
                    dot.addEventListener('click', function() {
                        goToHero(index);
                        resetHeroAutoplay();
                    });
                    dot.addEventListener('keydown', function(e) {
                        if (e.key === 'Enter' || e.key === ' ') {
                            // e.preventDefault();
                            goToHero(index);
                            resetHeroAutoplay();
                        }
                    });
                });
            }

            function startHeroAutoplay() {
                stopHeroAutoplay();
                heroInterval = setInterval(nextHero, 5000);
            }

            function stopHeroAutoplay() {
                if (heroInterval) {
                    clearInterval(heroInterval);
                    heroInterval = null;
                }
            }

            function resetHeroAutoplay() {
                startHeroAutoplay();
            }

            var heroContainer = document.querySelector('.hero-slider');
            if (heroContainer) {
                heroContainer.addEventListener('mouseenter', stopHeroAutoplay);
                heroContainer.addEventListener('mouseleave', startHeroAutoplay);
                heroContainer.addEventListener('focusin', stopHeroAutoplay);
                heroContainer.addEventListener('focusout', function(e) {
                    if (!this.contains(e.relatedTarget)) startHeroAutoplay();
                });
            }

            function initHeroSlider() {
                if (!getHeroSlides().length) {
                    stopHeroAutoplay();
                    return;
                }
                bindHeroDots();
                goToHero(0);
                startHeroAutoplay();
            }

            // Initialise for the pre-render state and re-initialise immediately
            // after the canonical renderer has produced the DB-driven slides.
            initHeroSlider();
            document.addEventListener('hero:rendered', initHeroSlider);

            // === COPYRIGHT YEAR ===
            var yearEl = document.getElementById('copyrightYear');
            if (yearEl) {
                yearEl.textContent = new Date().getFullYear();
            }

            // === UPDATE ACTIVE NAV LINK ===
            var currentPath = window.location.pathname;
            var navLinks = document.querySelectorAll('.nav-list a[href]');
            navLinks.forEach(function(link) {
                var href = link.getAttribute('href');
                if (href === currentPath || (href === 'index.html' && (currentPath === '/' || currentPath === '' || currentPath === '/index.html'))) {
                    link.classList.add('active');
                    link.setAttribute('aria-current', 'page');
                } else {
                    link.classList.remove('active');
                    link.removeAttribute('aria-current');
                }
            });

            // === DESKTOP DROPDOWN KEYBOARD ACCESS ===
            // On desktop, allow Enter/Space to toggle dropdown
            var desktopDropdownParents = document.querySelectorAll('.nav-list > li > a.dropdown-parent');
            desktopDropdownParents.forEach(function(link) {
                link.addEventListener('keydown', function(e) {
                    if (window.innerWidth > 768) {
                        if (e.key === 'Enter' || e.key === ' ') {
                            // e.preventDefault();
                            var parentLi = this.parentElement;
                            var isExpanded = parentLi.getAttribute('aria-expanded') === 'true';
                            parentLi.setAttribute('aria-expanded', String(!isExpanded));
                        }
                    }
                });
            });

            // Close dropdowns when clicking outside
            document.addEventListener('click', function(e) {
                if (window.innerWidth > 768) {
                    var dropdownParents = document.querySelectorAll('.nav-list > li[aria-expanded="true"]');
                    dropdownParents.forEach(function(parent) {
                        if (!parent.contains(e.target)) {
                            parent.setAttribute('aria-expanded', 'false');
                        }
                    });
                }
            });
        });
    </script>
